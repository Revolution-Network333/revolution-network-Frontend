import requests, time def scrape_residential(url, extract_fields): r = requests.post( "https://api.revolution-network.fr/api/enterprise/v1/jobs", headers={"x-api-key": "YOUR_KEY"}, json={ "type": "scrape_get", "params": { "url": url, "mode": "auto", "extract": extract_fields } } ) job_id = r.json()["job_id"] # polling jusqu'à "done" while True: res = requests.get(f".../{job_id}/result", ...) if res.json()["status"] == "done": return res.json()["output"] time.sleep(1)
